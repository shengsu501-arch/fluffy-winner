    <script>
        const music = document.getElementById('bgMusic');
        const statusText = document.getElementById('status');
        const progressBar = document.getElementById('progress');
        
        // 自动播放音乐
        function startMusic() {
            // 显示加载进度
            let progress = 0;
            const loadingInterval = setInterval(() => {
                progress += 10;
                progressBar.style.width = progress + '%';
                
                if (progress >= 90) {
                    clearInterval(loadingInterval);
                    statusText.textContent = "正在启动音乐播放器...";
                    
                    // 尝试自动播放
                    setTimeout(() => {
                        music.volume = 0.7; // 设置初始音量
                        
                        const playPromise = music.play();
                        
                        if (playPromise !== undefined) {
                            playPromise.then(() => {
                                // 播放成功
                                statusText.innerHTML = "✅ 音乐开始自动播放！";
                                progressBar.style.width = '100%';
                                progressBar.style.background = '#00ff88';
                                
                                // 更新音乐信息显示
                                setInterval(updateMusicInfo, 1000);
                            }).catch(error => {
                                // 自动播放被阻止
                                statusText.innerHTML = "🔇 自动播放被阻止<br>请<span style='color:#00ff88'>点击页面任意位置</span>播放音乐";
                                document.body.style.cursor = 'pointer';
                                
                                // 点击页面播放
                                document.body.onclick = function() {
                                    music.play();
                                    statusText.innerHTML = "✅ 音乐开始播放！";
                                    document.body.style.cursor = 'default';
                                    document.body.onclick = null;
                                };
                            });
                        }
                    }, 1000);
                }
            }, 200);
        }
        
        // 更新音乐信息
        function updateMusicInfo() {
            if (!music.paused) {
                const currentTime = formatTime(music.currentTime);
                const duration = formatTime(music.duration);
                statusText.innerHTML = `🎶 播放中 | ${currentTime} / ${duration}`;
            }
        }
        
        // 格式化时间
        function formatTime(seconds) {
            if (isNaN(seconds)) return "0:00";
            const mins = Math.floor(seconds / 60);
            const secs = Math.floor(seconds % 60);
            return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
        }
        
        // 控制函数
        function toggleMusic() {
            if (music.paused) {
                music.play();
                document.querySelector('button').innerHTML = '⏸️ 暂停';
            } else {
                music.pause();
                document.querySelector('button').innerHTML = '▶️ 播放';
            }
        }
        
        function restartMusic() {
            music.currentTime = 0;
            music.play();
            document.querySelector('button').innerHTML = '⏸️ 暂停';
        }
        
        function changeVolume(level) {
            music.volume = level;
            statusText.innerHTML = `音量设置为: ${Math.round(level * 100)}%`;
        }
        
        // 页面加载完成后开始
        window.onload = function() {
            setTimeout(startMusic, 1000);
        };
        
        // 监听音乐播放进度
        music.addEventListener('timeupdate', function() {
            if (music.duration) {
                const percent = (music.currentTime / music.duration) * 100;
                progressBar.style.width = percent + '%';
            }
        });
    </script>
